MkDocsSiteTemplate

data class MkDocsSiteTemplate(val rootPath: Path, val siteName: String, val docsDirName: String = MkDocsProject.DEFAULT_DOCS_DIR, val assetsDirName: String = MkDocsProject.DEFAULT_ASSETS_DIR, val siteDirName: String = MkDocsProject.DEFAULT_SITE_DIR, val features: List<MkDocsSiteFeature> = emptyList())
